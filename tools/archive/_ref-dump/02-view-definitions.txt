==== VIEW VIEW_barcord ====
def
CREATE VIEW dbo.VIEW_barcord  AS  SELECT  dbo.dm_wz.comm, ISNULL(dbo.dm_wz.cz, dbo.dm_wz.dm) AS dm, RTRIM(dbo.dm_wz.mc) AS mc, RTRIM(dbo.dm_wz.jldw)                      AS jldw, RTRIM(dbo.dm_wz.lb) AS lb, dbo.dm_wz.rkj, dbo.dm_wz.pict, RTRIM(dbo.dm_wz.ys) AS ys, RTRIM(dbo.dm_wz.cz)                      AS cz, RTRIM(dbo.dm_wz.gg) AS gg, dbo.dm_wz.bz, dbo.dm_wz.yxrq, dbo.dm_wz.cj, dbo.dm_wz.lsj, dbo.dm_wz.fxj,                      dbo.kucun.ckdm, dbo.kucun.lot_no, dbo.kucun.color, dbo.kucun.siz, SUM(dbo.kucun.sl) AS sl, SUM(dbo.kucun.total_je)                      AS total_je, 0 AS cost  FROM      dbo.dm_wz LEFT OUTER JOIN                     dbo.kucun ON dbo.dm_wz.comm = dbo.kucun.comm AND dbo.dm_wz.dm = dbo.kucun.wzdm  GROUP BY ISNULL(dbo.dm_wz.cz, dbo.dm_wz.dm), RTRIM(dbo.dm_wz.mc), RTRIM(dbo.dm_wz.jldw), RTRIM(dbo.dm_wz.lb),                      dbo.dm_wz.rkj, dbo.dm_wz.pict, RTRIM(dbo.dm_wz.ys), RTRIM(dbo.dm_wz.cz), RTRIM(dbo.dm_wz.gg), dbo.dm_wz.bz,                      dbo.dm_wz.yxrq, dbo.dm_wz.cj, dbo.dm_wz.lsj, dbo.dm_wz.fxj, dbo.kucun.lot_no, dbo.kucun.color, dbo.kucun.siz,                      dbo.kucun.ckdm, dbo.dm_wz.comm  
-- 1 rows

==== VIEW View_BOM ====
def
CREATE VIEW dbo.View_BOM  AS  SELECT  TOP (100) PERCENT dbo.mate.comm, dbo.mate.m_no, dbo.mate.m_date, dbo.mate.wzdm1, dbo.dm_wz.mc, dbo.dm_wz.gg,                      dbo.mate.lot_no1, dbo.mate.color1, dbo.mate.siz1, dbo.mate.ywdm, dbo.dm_ywy.mc AS ywmc, dbo.mate.bz  FROM      dbo.mate LEFT OUTER JOIN                     dbo.dm_ywy ON dbo.mate.comm = dbo.dm_ywy.comm AND dbo.mate.ywdm = dbo.dm_ywy.dm LEFT OUTER JOIN                     dbo.dm_wz ON dbo.mate.comm = dbo.dm_wz.comm AND dbo.mate.wzdm1 = dbo.dm_wz.dm  GROUP BY dbo.mate.m_no, dbo.mate.m_date, dbo.mate.wzdm1, dbo.dm_wz.mc, dbo.dm_wz.gg, dbo.mate.lot_no1, dbo.mate.color1,                      dbo.mate.siz1, dbo.mate.ywdm, dbo.dm_ywy.mc, dbo.mate.bz, dbo.mate.comm  ORDER BY dbo.mate.m_no DESC  
-- 1 rows

==== VIEW VIEW_CK ====
def
CREATE VIEW dbo.VIEW_CK  AS  SELECT  dbo.outh.comm, dbo.outh.wzdm, dbo.outh.ckdm, dbo.outh.lot_no, dbo.outh.color, dbo.outh.siz, SUM(dbo.outh.sl) AS rkl, MAX(dbo.outh.out_date) AS indate,                  SUM(dbo.outh.ddwsl) AS sl2, dbo.outh.bz1, dbo.outh.outh_no AS inh_no, dbo.outh.khdm AS gfdm, dbo.outh.ywman, dbo.users.userna,                  dbo.dm_ck.mc AS ckmc, dbo.outh.od_no AS od_no2, dbo.outh.od_xc AS od_xc2, dbo.outh.o_zk AS i_zk, dbo.outh.o_zke AS i_zke, dbo.outh.total_m,                  dbo.outh.tax, dbo.outh.tax_dm, dbo.outh.tax_sm, dbo.outh.bz, dbo.outh.qr, dbo.outh.id + 50000000000 AS id  FROM     dbo.outh LEFT OUTER JOIN                 dbo.dm_ck ON dbo.outh.comm = dbo.dm_ck.comm AND dbo.outh.ckdm = dbo.dm_ck.dm LEFT OUTER JOIN                 dbo.users ON dbo.outh.comm = dbo.users.comm AND dbo.outh.ywman = dbo.users.id  WHERE   (CONVERT(varchar(6), dbo.outh.out_date, 112) = CONVERT(varchar(6), GETDATE(), 112)) AND (dbo.outh.qr = N'Y')  GROUP BY dbo.outh.wzdm, dbo.outh.ckdm, dbo.outh.lot_no, dbo.outh.color, dbo.outh.siz, dbo.outh.comm, dbo.outh.bz1, dbo.outh.outh_no, dbo.outh.khdm,                  dbo.outh.ywman, dbo.users.userna, dbo.dm_ck.mc, dbo.outh.od_no, dbo.outh.od_xc, dbo.outh.o_zk, dbo.outh.o_zke, dbo.outh.total_m, dbo.outh.tax,                  dbo.outh.tax_dm, dbo.outh.tax_sm, dbo.outh.bz, dbo.outh.qr, dbo.outh.id + 50000000000  
-- 1 rows
