# 库存三表结构（inh 入库流水 / outh 出库流水 / kucun 结存）实施计划

**目标：** 把库存从「`kucun` 一张表（只有累计入/累计出/余量三个数）」改为「**两张流水表 + 一张结存表**」，`kucun` 退为结存缓存，**流水为唯一真源**。

**架构：** 复用遗留表名 `inh`（入库流水）/ `outh`（出库流水）——按 MES 语义**清空重建列**（老结构是纺织行业的面料进货单，115/126 列、数量单位是"米/公斤"，与本系统按个/支的滤芯制造语义冲突）。8 类出入库单据**审核时**由 `StockLedgerService` 在同一事务里写流水（`kucun` 同时更新，作为结存缓存）；`v_stock_movement` 从「8 张单据表 UNION」改为「读 `inh`/`outh`」，从而**根治报表 65 秒超时**；新增「结存 == Σ流水」启动对账自检。

**技术栈：** Java 25 / Spring Boot / JdbcTemplate · SQL Server（HSDZ_MES + HSDZ_MES_TEST 两账套）· Vue3 前端（本计划几乎不动前端）· Node 探针（`tools/archive/_probe-*.cjs`）

**规格：** [库存表清单.md](../development/库存表清单.md) 的 §5「需要拍板的 4 件事」第 1 条（期初要不要并进报表，本条按"要"做）与第 4 条（审核口径统一成单一真源），以及用户 2026-09-30 口径：**直接清空重建** + **kucun 当结存缓存、流水为准**。

## 全局约束

- 一切库变更**两个账套都执行**（先 `HSDZ_MES` 正式、后 `HSDZ_MES_TEST` 测试），脚本必须幂等
- 新增表/列必须同脚本写 `MS_Description` 中文注明；查注明时按 `class=1` 过滤（索引级会撞号）
- 数据键一律中文（ADR-0001）；新增字段标签/面板名必须同时交付 `yj_translation` 译名（至少 en）
- 迁移脚本要在 `tools/db-migrations.txt` 登记；收尾跑 `DbSync sync` 到「执行 0、失败 0」
- 库存主链路每改一步都要探针验证；**不许用"构建通过"代替功能验证**
- 提交格式 `<type>: <中文摘要>`，单任务单提交
- 表前缀按 `数据库表清单.md` §0 选：**本计划沿用 `inh`/`outh` 这两个既有名**（用户明确要求"利用留下来的表"），**不新增 `bd_`/`bl_` 表**

```java
package com.yinjia.mes.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;

/**
 * 库存流水读写(inh 入库 / outh 出库)。2026-09-30 用户口径:库存改为三表,
 * kucun 退为**结存缓存**、流水为唯一真源。
 *
 * 与 {@link StockLedgerService} 的分工:
 *   · StockLedgerService 负责**结存**(kucun.yl/rkl/ckl)与业务守卫(现存量不足、仓库档案必须存在);
 *   · 本类负责**流水**(inh/outh)的写入与红冲。
 * 两者由 StockLedgerService 在**同一事务**内先后调用 —— 一旦有一边失败,整笔回滚,
 * 不会出现"结存动了流水没动"(这正是静态检查项「结存 == Σ流水」要防的)。
 *
 * 幂等:(src, rid) 上有唯一索引,同一单据行重复审核不会重复记流水。
 */
@Service
public class StockFlowService {

    private static final Logger log = LoggerFactory.getLogger(StockFlowService.class);

    private final JdbcTemplate jdbc;

    public StockFlowService(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    /** 入库面板(src 1..4)。 */
    private static int inSrc(String panelCode) {
        return switch (panelCode) {
            case "PURCHASE_IN" -> 1;
            case "FINISH_IN" -> 2;
            case "OTHER_IN" -> 3;
            case "OUTSOURCE_IN" -> 4;
            default -> -1;
        };
    }

    /** 出库面板(src 5..8)。 */
    private static int outSrc(String panelCode) {
        return switch (panelCode) {
            case "SALE_OUT" -> 5;
            case "MATERIAL_OUT" -> 6;
            case "OTHER_OUT" -> 7;
            case "OUTSOURCE_ISSUE" -> 8;
            default -> -1;
        };
    }

    /**
     * 写流水。
     *
     * @param rows     已由 StockLedgerService.loadRows 取好的单据行(键:code/lot/qty/price/
     *                 行仓库/头仓库/行仓库名称/行仓库编码/头仓库编码/rid/单据编号/单据日期/往来单位/经手人)
     * @param forward  true=审核(插入流水) / false=弃审(红冲)
     * @return 影响行数
     */
    public int postFlow(String panelCode, String no, List<Map<String, Object>> rows, boolean forward, String user) {
        int inS = inSrc(panelCode), outS = outSrc(panelCode);
        if (inS < 0 && outS < 0) return 0;
        if (rows == null || rows.isEmpty()) return 0;
        int n = 0;
        for (Map<String, Object> r : rows) {
            Object rid = r.get("rid");
            if (rid == null) continue;                       // 没有来源行 id 就不记流水(无法幂等)
            String tbl = inS > 0 ? "inh" : "outh";
            double qty = dbl(r.get("qty"));
            if (qty == 0) continue;
            Integer exists = jdbc.queryForObject(
                    "SELECT COUNT(*) FROM " + tbl + " WHERE src = ? AND rid = ?", Integer.class,
                    inS > 0 ? inS : outS, rid);
            if (forward) {
                if (exists != null && exists > 0) continue;   // 幂等:已记过
                if (inS > 0) {
                    jdbc.update("INSERT INTO inh (src, rid, 单据编号, 单据类型, 单据日期, 物料编码, 物料名称,"
                                    + " 规格型号, 计量单位, 仓库编码, 仓库名称, 批号, 数量, 单价, 金额, 往来单位, 经手人,"
                                    + " asp_user1, asp_time1, asp_cancel)"
                                    + " VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,GETDATE(),'N')",
                            inS, rid, no, str(r.get("单据类型")), r.get("单据日期"), str(r.get("code")),
                            str(r.get("name")), str(r.get("spec")), str(r.get("uom")),
                            str(r.get("仓库编码")), str(r.get("仓库名称")), str(r.get("lot")),
                            qty, dbl(r.get("price")), dbl(r.get("金额")),
                            str(r.get("往来单位")), str(r.get("经手人")), user);
                } else {
                    jdbc.update("INSERT INTO outh (src, rid, 单据编号, 单据类型, 单据日期, 物料编码, 物料名称,"
                                    + " 规格型号, 计量单位, 仓库编码, 仓库名称, 批号, 数量, 单据金额, 往来单位, 经手人,"
                                    + " asp_user1, asp_time1, asp_cancel)"
                                    + " VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,GETDATE(),'N')",
                            outS, rid, no, str(r.get("单据类型")), r.get("单据日期"), str(r.get("code")),
                            str(r.get("name")), str(r.get("spec")), str(r.get("uom")),
                            str(r.get("仓库编码")), str(r.get("仓库名称")), str(r.get("lot")),
                            qty, dbl(r.get("金额")),
                            str(r.get("往来单位")), str(r.get("经手人")), user);
                }
                n++;
            } else {
                // 弃审:软删(asp_cancel='Y'),保留痕迹;不物理删除
                int m = jdbc.update("UPDATE " + tbl + " SET asp_cancel = 'Y', asp_user4 = ?, asp_time4 = GETDATE()"
                        + " WHERE src = ? AND rid = ? AND ISNULL(asp_cancel,'N') <> 'Y'",
                        user, inS > 0 ? inS : outS, rid);
                n += m;
            }
        }
        if (n > 0) log.info("[库存流水] {} {} {} {} 行", panelCode, no, forward ? "写入" : "红冲", n);
        return n;
    }

    /** 该单据在三键上的流水净额,供「结存 == Σ流水」对账用。 */
    private static String str(Object o) { return o == null ? null : String.valueOf(o); }
    private static double dbl(Object o) { return o == null ? 0d : Double.parseDouble(String.valueOf(o)); }
}
```

/* 口径变更(2026-09-30):
   原:直接 UNION 8 张 bd_/bl_ 单据表,每次查询要现场拼 8 路 ⇒ 实测缓存计划 65,021 ms(前端 15s 超时)。

CREATE VIEW dbo.v_stock_movement AS
SELECT m.src, m.rid, m.单据日期, m.单据类型, m.单据编号, m.业务类型,
